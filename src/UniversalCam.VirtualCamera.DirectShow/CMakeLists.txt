cmake_minimum_required(VERSION 3.20)
project(UniversalCamVCam LANGUAGES CXX)
set(BASECLASSES_DIR "" CACHE PATH "Microsoft DirectShow baseclasses")
file(GLOB BASE_SOURCES "${BASECLASSES_DIR}/*.cpp")
add_library(UniversalCamVCam SHARED ${BASE_SOURCES} pch.cpp UCamCapturePin.cpp UCamCaptureFilter.cpp dllmain.cpp UCamVCam.def)
target_include_directories(UniversalCamVCam PRIVATE "${BASECLASSES_DIR}" .)
target_compile_definitions(UniversalCamVCam PRIVATE UNICODE _UNICODE WIN32 _WINDOWS _USRDLL)
target_compile_options(UniversalCamVCam PRIVATE /EHsc /wd4996)
target_link_libraries(UniversalCamVCam PRIVATE strmiids winmm msacm32 ole32 oleaut32 uuid advapi32 user32 gdi32)
